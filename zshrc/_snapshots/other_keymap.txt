Keymap: other_keymap.zsh

  `  (1) <2>  3   4   5   |   6   7  <8>  9  <0>  [   ] 
      '   ,   .  <p> <y>  |  <f>  g  (c) <r>  l   /   =   \ 
     <a> (o) <e> (u) (i)  |  (d)  h  <t> <n> <s>  - 
      ;  <q> <j> (k) (x)  |  <b> <m> <w> <v>  z 

  `<>` initials have only one key mapping
  `()` initials have multiple key mappings

Keymap Usage

  $ o                               # Show this keymap
  $ o <regex>                       # Search this keymap

  $ o.<key>                         # Key takes no variable
  $ o.<key> <var>                   # Key takes one variable
  $ o.<key> <var>?                  # Key takes zero or one variable
  $ o.<key> <var>*                  # Key takes zero or more variables
  $ o.<key> (1-10)                  # Key takes a value from inside list
  $ o.<key> (^|)?                   # Key can be piped to: ... | o.<key>

     ^                              # The `.` is for visual clarity
                                    # Omit it when invoking a key

Keymap List

  $ o <path>                        # Open path in Finder
  $ o <url>                         # Open url in browser
  $ o.o                             # Open current dir in Finder
  $ o.e <path>?                     # Open IntelliJ (Default: Repo root / CWD)
  $ o.m <path>?                     # Open TextMate (Default: CWD)

  $ o.a                             # Stay awake
  $ o.s                             # Sleep

  $ o.cc                            # Copy command
  $ o.c                             # Copy output
  $ o.v                             # Speak pasteboard
  $ o.y                             # Alias for `pbcopy`
  $ o.p                             # Alias for `pbpaste`

  $ o.1 (^|)?                       # Save pasteboard value to `diff.1.txt`
  $ o.2 (^|)?                       # Save pasteboard value to `diff.2.txt`
  $ o.u <file 1>? <file 2>?         # Unified diff
  $ o.uu <file 1>? <file 2>?        # Side by side diff
  $ o.12                            # Open diff files in TextMate
  $ o.0                             # Empty `diff.1.txt` and `diff.2.txt`

  $ o.i <index> <file>? (^|)?       # CSV: Keep column by index
  $ o.id <index> <file>? (^|)?      # CSV: Delete column by index
  $ o.ii <index> <file>? (^|)?      # CSV: Sort lines by column
  $ o.ix <i1> <i2> <file>? (^|)?    # CSV: Swap columns
  $ o.x <file 1>? <file 2>?         # CSV: Keep lines matching 1st column
  $ o.xx <file 1>? <file 2>?        # CSV: Delete lines matching 1st column

  $ o.t <command>                   # Time a command
  $ o.w <seconds>? <command>        # Watch a command (Default: 1s)
  $ o.f <start> <finish> (~~)       # Run a command sequence in foreground
  $ o.b <start> <finish> (~~)       # Run a command sequence in background

  $ o.k                             # Clear terminal
  $ o.kk                            # Show archived terminal outputs
  $ o.kc                            # Clear archived terminal outputs

  $ o.8                             # Use Java 8
  $ o.d <url>?                      # DNS dig (Default: Pasteboard)
  $ o.df                            # DNS flush
  $ o.j <url> <regex>? <num lines>? # Curl json endpoint
  $ o.n                             # Encrypt pasteboard + print decrypt cmd
  $ o.q '<sql>'? (^|)?              # Format a sql query (Default: Pasteboard)
  $ o.r <before> <after>            # Rename files in current directory
