Keymap: aws_keymap.zsh

  `   1   2   3   4   5   |   6   7   8   9   0   [   ] 
      '   ,   .  <p>  y   |   f   g  (c)  r   l   /   =   \ 
     (a) (o) (e) <u>  i   |   d  <h> <t> (n) (s) <->
      ;  (q)  j   k   x   |   b  (m) (w)  v   z 

  `<>` initials have only one key mapping
  `()` initials have multiple key mappings

Keymap Usage

  $ s                           # Show this keymap
  $ s <regex>                   # Search this keymap

  $ s.<key>                     # Key takes no variable
  $ s.<key> <var>               # Key takes one variable
  $ s.<key> <var>?              # Key takes zero or one variable
  $ s.<key> <var>*              # Key takes zero or more variables
  $ s.<key> (1-10)              # Key takes a value from inside list
  $ s.<key> (^|)?               # Key can be piped to: ... | s.<key>

     ^                          # The `.` is for visual clarity
                                # Omit it when invoking a key

Keymap List

  $ s.s <match>* <-mismatch>*   # List Britive roles & filter
  $ s <role>                    # Assume specified role

  $ s.e1                        # Use us-east-1 region
  $ s.e2                        # Use us-east-2 region
  $ s.w2                        # Use us-west-2 region
  $ s.c1                        # Use eu-central-1 region

  $ s.a <name>                  # ASG search
  $ s.aa <asg id>               # ASG open in new tab
  $ s.e <name>                  # EC2 search
  $ s.ee <ec2 id OR ip OR name> # EC2 open in new tab
  $ s.o                         # SSM start session with `sudo -i`
  $ s.oc                        # SSM start session with command
  $ s.oo                        # SSM start session
  $ s.-                         # SSM: Copy history bindings

  $ s.c <name>                  # Code Pipeline search
  $ s.cc <name>                 # Code Pipeline get latest status

  $ s.m <name>                  # Secret Manager search
  $ s.mg <name> <version>?      # Secret Manager get by version
  $ s.md <name>                 # Secret Manager delete

  $ s.n <name>                  # SNS search
  $ s.nn <topic arn>            # SNS open in new tab

  $ s.q <name>                  # SQS search
  $ s.qq <queue url>            # SQS open in new tab
  $ s.qg <queue url>            # SQS get stats
  $ s.qr <queue url> <count>?   # SQS receive message
  $ s.qp <queue url>            # SQS purge

  $ s.p <name>                  # Parameter Store get latest version

  $ s.t <message>               # STS decode

  $ s.w <id, ip, name>?         # Which acct or EC2? (Default: Pasteboard)

  $ s.h                         # (Reserved: Shell)
  $ s.u                         # (Reserved: Switch user)
