Keymap: github_keymap.zsh

  `   1   2   3   4   5   |   6   7   8   9   0   [   ] 
      '   ,   .  <p>  y   |   f  (g) <c> (r)  l   /   =   \ 
     <a> (o)  e  <u>  i   |  <d> <h>  t  <n>  s   - 
      ;   q   j   k   x   |  (b)  m   w   v   z 

  `<>` initials have only one key mapping
  `()` initials have multiple key mappings

Keymap Usage

  $ h                         # Show this keymap
  $ h <regex>                 # Search this keymap

  $ h.<key>                   # Key takes no variable
  $ h.<key> <var>             # Key takes one variable
  $ h.<key> <var>?            # Key takes zero or one variable
  $ h.<key> <var>*            # Key takes zero or more variables
  $ h.<key> (1-10)            # Key takes a value from inside list
  $ h.<key> (^|)?             # Key can be piped to: ... | h.<key>

     ^                        # The `.` is for visual clarity
                              # Omit it when invoking a key

Keymap List

  $ h.h <match>* <-mismatch>* # List ~/GitHub, `cd` if 1 match
  $ h.r <match>* <-mismatch>* # List remote repos & filter
  $ h.rr                      # Save copy of remote repos

  $ h.a <repo>?               # Open GitHub app (Default: Current repo)
  $ h.n                       # Open tab to new PR
  $ h.g (^|)?                 # Open tab to new gist
  $ h.gg                      # Open tab to create gist

  $ h.o <path>?               # Open tab to path at SHA (Default: CWD)
  $ h.oo <path>?              # Open tab to path at main (Default: CWD)
  $ h.b                       # Open tab to branches
  $ h.p <pr>?                 # Open tab to PRs (or a specific PR)
  $ h.c <sha>?                # Open tab to commit (Default: Pasteboard)

  $ h.url                     # Remote url
  $ h.domain                  # Remote domain
  $ h.org                     # Org name
  $ h.repo                    # Repo name
  $ h.branch                  # Branch name
