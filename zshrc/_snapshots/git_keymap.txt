Keymap: git_keymap.zsh

  `   1   2   3   4   5   |   6   7   8   9   0   [   ] 
      '   ,   .  (p)  y   |  (f) (g) (c) (r) (l)  /   =   \ 
     <a> <o> (e) <u> <i>  |  <d> <h> (t) <n> <s>  - 
      ;   q   j   k   x   |  (b) (m) (w)  v  (z)

  `<>` initials have only one key mapping
  `()` initials have multiple key mappings

Keymap Usage

  $ g               # Show this keymap
  $ g <regex>       # Search this keymap

  $ g.<key>         # Key takes no variable
  $ g.<key> <var>   # Key takes one variable
  $ g.<key> <var>?  # Key takes zero or one variable
  $ g.<key> <var>*  # Key takes zero or more variables
  $ g.<key> (1-10)  # Key takes a value from inside list
  $ g.<key> (^|)?   # Key can be piped to: ... | g.<key>

     ^              # The `.` is for visual clarity
                    # Omit it when invoking a key

Keymap List

  $ g <branch>      # Checkout branch
  $ g.g             # Checkout latest `main`
  $ g.n <branch>    # Checkout latest `main` & create branch

  $ g.b             # List local branches
  $ g.bb            # Delete merged branches
  $ g.bd <branch>   # Delete specified branch

  $ g.i             # Status
  $ g.d             # Diff
  $ g.s <message>?  # Stash
  $ g.a <index>?    # Apply a stash (Default: Latest)
  $ g.l             # List stashes
  $ g.lc            # Clear stashes

  $ g.w             # Write a new commit
  $ g.ww            # Write a new commit, allow empty
  $ g.m             # Amend previous commit, no edit
  $ g.mm            # Amend previous commit

  $ g.u <number>?   # Undo last N commits (Default: 1)
  $ g.z <number>?   # Discard changes & N commits (Default: 0)
  $ g.zz            # Discard changes & 1 commit

  $ g.e             # List last 20 commits
  $ g.e <sha>       # Create fixup commit
  $ g.ee (<N>,u,m)? # Rebase (Default: origin/main)
  $ g.ec            # Rebase continue
  $ g.ea            # Rebase abort

  $ g.c <sha>?      # Cherry pick (Default: Pasteboard)
  $ g.cc            # Cherry pick continue
  $ g.ca            # Cherry pick abort

  $ g.t <sha>?      # Revert commit (Default: Pasteboard)
  $ g.tc            # Revert continue
  $ g.ta            # Revert abort

  $ g.pp            # Pull
  $ g.p             # Push
  $ g.f             # Force push with lease
  $ g.ff            # Force push

  $ g.r <match>*    # List commits & filter
  $ g.rr <match>*   # List first-parent commits & filter

  $ g.h             # (Reserved: GitHub CLI)
  $ g.o             # (Reserved: Go language)
